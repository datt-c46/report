\section{Kiến trúc tổng thể}\label{sec:architecture}
\subsection{Các thành phần}
Hệ thống gồm hai phần mềm do nhóm phát triển (Sơ đồ~\ref{dia:architecture}):
\begin{itemize}
  \item \textbf{Ứng dụng desktop} \textit{voice-calendar-agent} (Electron): tiến trình giao diện thu âm, chạy VAD, hiển thị hội thoại và phát âm thanh; tiến trình chính giữ phiên đăng nhập và là nơi duy nhất gọi backend.
  \item \textbf{Dịch vụ} \textit{voice-backend} (Fastify): cung cấp API REST dưới tiền tố \ma{/api/v1} và một kênh WebSocket; gồm bảy mô-đun \textit{health, auth, users, calendar, agent, voice, realtime}.
\end{itemize}
Backend kết nối tới PostgreSQL và các dịch vụ ngoài: Google OAuth, Google Calendar API, Gemini API; Piper được chạy như tiến trình con trên chính máy chủ. Trong sơ đồ, \textit{lớp nền} là các plugin dùng chung cho mọi tuyến; các \textit{bộ nối} tới dịch vụ ngoài nằm trong từng mô-đun và đều ẩn sau giao diện trừu tượng.

\subsection{Nguyên tắc thiết kế}
\begin{itemize}
  \item \textbf{Client mỏng:} ứng dụng desktop không chứa khóa API, client secret hay token Google; mọi thao tác đặc quyền nằm ở backend.
  \item \textbf{Tách tầng rõ ràng trong ứng dụng:} tiến trình giao diện không bao giờ nhìn thấy token; mọi lời gọi đi theo đường renderer → IPC → main → backend.
  \item \textbf{Mô-đun hóa backend:} mỗi mô-đun công bố một giao diện dịch vụ; các mô-đun được nối với nhau tại một nơi duy nhất (container) theo thứ tự phụ thuộc \textit{auth → users → calendar → agent → voice}.
  \item \textbf{Nhà cung cấp thay thế được:} STT, TTS, mô hình ngôn ngữ, cổng OAuth và API lịch đều là giao diện; bản hiện thực thật và bản giả lập cho kiểm thử dùng chung hợp đồng.
  \item \textbf{Thất bại tường minh:} thành phần không khả dụng trả lỗi có mã rõ ràng (ví dụ \ma{STT_UNAVAILABLE}) thay vì âm thầm trả kết quả giả.
\end{itemize}

\subsection{Quản lý phiên đăng nhập phía ứng dụng}
Lớp \textit{BackendClient} ở tiến trình chính là nơi duy nhất giữ cặp access token/refresh token. Các điểm thiết kế đáng chú ý:
\begin{itemize}
  \item Access token được làm mới trước khi hết hạn 30 giây. Các yêu cầu đồng thời cùng cần làm mới được gộp thành \textit{một} lời gọi (single-flight), vì máy chủ coi việc dùng lại refresh token là dấu hiệu bị đánh cắp và sẽ thu hồi cả chuỗi phiên.
  \item Phiên được lưu xuống đĩa dưới dạng đã mã hóa bằng \ma{safeStorage}, ghi tệp theo kiểu nguyên tử (ghi tệp tạm rồi đổi tên). Nếu hệ điều hành không cung cấp được kho khóa an toàn (kể cả trường hợp Linux rơi về chế độ \ma{basic_text}), ứng dụng \textit{không} ghi token dạng rõ mà chỉ giữ phiên trong bộ nhớ; người dùng đăng nhập lại ở lần mở sau.
  \item Tệp cấu hình \ma{.env} của ứng dụng chỉ được đọc hai khóa \ma{VOICE_BACKEND_URL} và \ma{VOICE_BACKEND_DEV_EMAIL}; mọi khóa khác bị bỏ qua.
  \item Cửa sổ chỉ hiển thị giao diện đóng gói sẵn; mọi điều hướng khác và liên kết ngoài được chuyển cho trình duyệt hệ thống, và chỉ với giao thức http/https. Quyền thiết bị chỉ cấp cho microphone.
\end{itemize}

\subsection{Các quy ước chung}
\begin{itemize}
  \item \textbf{Cấu hình:} mọi biến môi trường được kiểm tra bằng lược đồ khi khởi động; cấu hình sai làm dịch vụ dừng ngay và liệt kê toàn bộ lỗi. Một số cấu hình nguy hiểm bị từ chối ở chế độ production (bật đăng nhập phát triển, dùng STT giả lập).
  \item \textbf{Lỗi:} mọi phản hồi lỗi có cùng một dạng phong bì gồm mã lỗi, thông điệp, chi tiết và mã yêu cầu; có 17 mã lỗi (ví dụ \ma{VALIDATION_ERROR}, \ma{UNAUTHORIZED}, \ma{GOOGLE_REAUTH_REQUIRED}, \ma{RATE_LIMITED}, \ma{STT_UNAVAILABLE}). Thông điệp của lỗi 5xx luôn chung chung, không lộ thông tin nội bộ hay thông điệp gốc của dịch vụ ngoài.
  \item \textbf{Thời gian:} mọi mốc thời gian trao đổi qua API ở dạng ISO-8601 có độ lệch múi giờ; múi giờ hiệu lực của một yêu cầu được xác định theo thứ tự: tham số của yêu cầu → cài đặt người dùng → mặc định của máy chủ. Logic nghiệp vụ không gọi trực tiếp đồng hồ hệ thống mà nhận một đối tượng đồng hồ, cho phép kiểm thử với thời điểm cố định.
  \item \textbf{Cô lập dữ liệu:} mọi truy vấn gắn với mã người dùng; truy cập tài nguyên của người khác trả về 404 để không tiết lộ sự tồn tại. Các tuyến cần đăng nhập trả 401 trước khi kiểm tra nội dung yêu cầu.
  \item \textbf{Giới hạn tần suất:} giới hạn toàn cục theo người dùng hoặc theo IP, cộng thêm giới hạn riêng cho nhóm tuyến xác thực, giọng nói và lệnh agent (mục~\ref{sec:security-design}).
\end{itemize}

\subsection{Đăng nhập Google với PKCE ở hai chặng}

Các quy tắc kiểm tra chính:
\begin{itemize}
  \item \ma{redirect_uri} của ứng dụng phải thuộc danh sách cho phép hoặc là địa chỉ loopback (\ma{127.0.0.1}, \ma{localhost}, \ma{[::1]}); URL chứa thông tin đăng nhập, fragment hoặc cách viết IP bất thường bị từ chối.
  \item \textit{state} dùng một lần; mã đăng nhập dùng một lần, gắn với \ma{redirect_uri} và \ma{code_challenge} ban đầu; xuất trình sai \textit{verifier} vẫn làm mã bị hủy.
  \item Email Google chưa xác minh bị từ chối; không gộp vào tài khoản đã liên kết với một tài khoản Google khác.
  \item Chỉ xin bốn quyền: \ma{openid}, \ma{email}, \ma{profile} và \ma{calendar.events}. Lịch chỉ chuyển sang chế độ Google khi Google thực sự cấp \ma{calendar.events} và trả refresh token; nếu không, đăng nhập vẫn thành công nhưng người dùng tiếp tục dùng lịch nội bộ.
\end{itemize}

\subsection{Phiên làm việc và thông tin xác thực Google}
\begin{itemize}
  \item \textbf{Access token} là JWT ký HS256, mặc định có hạn 15 phút, mang mã người dùng, nơi phát hành và đối tượng nhận; mỗi yêu cầu đều kiểm tra người dùng còn tồn tại.
  \item \textbf{Refresh token} là chuỗi ngẫu nhiên, chỉ lưu dạng băm, mặc định có hạn 30 ngày và được xoay vòng mỗi lần dùng. Dùng lại một token đã xoay hoặc đã thu hồi — kể cả hai yêu cầu làm mới song song — bị coi là tái sử dụng và làm thu hồi toàn bộ chuỗi phiên.
  \item \textbf{Token Google} được mã hóa AES-256-GCM với vectơ khởi tạo ngẫu nhiên cho mỗi lần mã hóa. Access token Google được làm mới khi còn dưới 60 giây; các yêu cầu đồng thời của cùng người dùng dùng chung một lần làm mới. Khi Google trả \ma{invalid_grant}, thông tin xác thực được đánh dấu thu hồi và mọi thao tác lịch trả mã \ma{GOOGLE_REAUTH_REQUIRED} cho đến khi người dùng đăng nhập Google lại.
\end{itemize}
Ứng dụng phân biệt hai loại 401: \ma{UNAUTHORIZED} (làm mới token phiên rồi thử lại) và \ma{GOOGLE_REAUTH_REQUIRED} (phải đăng nhập Google lại; làm mới token phiên không giải quyết được).

\section{Thiết kế phân hệ lịch}
Mọi thao tác lịch đi qua giao diện \textit{ICalendarService} gồm sáu phép: liệt kê, đọc, tạo, cập nhật, xóa sự kiện và tính khoảng bận. Một bộ phân giải chọn bản hiện thực theo người dùng:
\begin{itemize}
  \item \textbf{Dịch vụ Google:} thao tác trên lịch \ma{primary}; đọc theo trang (tối đa 250 sự kiện mỗi trang) với sự kiện lặp được trải thành từng lần; mỗi yêu cầu có hạn 10 giây. Lỗi 429, 5xx và lỗi 403 do vượt hạn mức được thử lại tối đa ba lần với thời gian chờ tăng theo lũy thừa hai cộng nhiễu ngẫu nhiên. Lỗi 401 khiến token được làm mới một lần và thao tác chạy lại; lỗi quyền dẫn tới \ma{GOOGLE_REAUTH_REQUIRED}. Thao tác tạo dùng mã sự kiện do backend chọn trước, nên khi thử lại mà Google trả “đã tồn tại” thì sự kiện đã lưu được trả về thay vì tạo trùng.
  \item \textbf{Dịch vụ nội bộ:} lưu trong bảng \ma{local_events}; tìm kiếm không phân biệt hoa thường trên tiêu đề, mô tả, địa điểm.
\end{itemize}
Không có cơ chế âm thầm chuyển từ lịch Google sang lịch nội bộ: người dùng đã chọn Google mà mất quyền sẽ nhận lỗi yêu cầu đăng nhập lại, tránh tình trạng sự kiện bị ghi vào nơi người dùng không ngờ tới.

Các quy tắc thời gian của sự kiện được tập trung ở một nơi: thời điểm kết thúc phải sau thời điểm bắt đầu; sự kiện cả ngày dùng ngày kết thúc không bao gồm; chỉ đổi giờ bắt đầu thì giữ nguyên thời lượng; chuyển đổi qua lại giữa sự kiện cả ngày và sự kiện có giờ theo quy tắc xác định. Phép tính khoảng bận gộp các khoảng chồng lấn hoặc liền kề, bỏ qua sự kiện đã hủy và sự kiện đánh dấu “rảnh”.

Ba thành phần của chế độ lai:
\begin{itemize}
  \item \textbf{Bộ trích xuất LLM} gửi cho Gemini một lời nhắc hệ thống mô tả các ý định và quy tắc thời gian, kèm thời điểm hiện tại, thứ trong tuần (tiếng Việt và tiếng Anh) và múi giờ. Câu lệnh của người dùng được đưa vào như \textit{dữ liệu} (chuỗi JSON), kèm chỉ dẫn không làm theo bất kỳ yêu cầu nào nằm trong câu — một biện pháp giảm rủi ro chèn lệnh (prompt injection). Mô hình chỉ trả về ngày và giờ địa phương dạng \ma{YYYY-MM-DD} và \ma{HH:mm}; backend tự dựng lại thời điểm có múi giờ và đối chiếu với bộ phân tích tất định. Mô hình dự phòng chỉ được thử khi gặp lỗi 429, 404 hoặc 5xx.
  \item \textbf{Bộ luật} gồm bước chuẩn hóa văn bản, từ điển từ khóa cho từng ý định (so khớp theo ranh giới từ, xử lý cả cách gõ có dấu và không dấu), ngữ pháp thời gian và bước tách các trường. Câu phủ định (“đừng xóa”) không bị hiểu thành lệnh xóa.
  \item \textbf{Lời xác nhận ngắn} (tối đa sáu từ, ví dụ “có”, “ok”, “xóa đi”, “không”, “thôi”, “yes”, “cancel”) khi đang có thao tác chờ được quyết định trực tiếp bằng luật, không gọi LLM: nhanh, chính xác và không tốn hạn mức.
\end{itemize}
Nhật ký chỉ ghi lý do thất bại và mã trạng thái HTTP của lời gọi LLM, không ghi câu lệnh của người dùng.

\subsection{Suy luận ngày giờ cho tiếng Việt}
Ngữ pháp thời gian quét câu lệnh thành các \textit{thẻ thời gian} (ngày tuyệt đối, ngày tương đối, thứ trong tuần, khoảng tuần/tháng, giờ, buổi, thời lượng) rồi ghép chúng lại. Các quy tắc chính:
\begin{itemize}
  \item Một con số chỉ được coi là giờ khi đi kèm dấu hiệu tường minh: “lúc 3”, “3 giờ”, “15h”, “3h30”, “9:30”, “3 giờ rưỡi”, “3 giờ kém 15”, “3 chiều”. Số đứng sau các danh từ như \textit{thứ, ngày, mùng, tháng, phòng, tầng, số} không bao giờ là giờ.
  \item Nhận diện ngày: “ngày 20/11”, “20 tháng 11 năm 2026”, “mùng 5”; “hôm nay”, “ngày mai”, “mốt”; “thứ sáu”, “thứ sáu tuần sau”; “tuần này”, “cuối tuần”, “tháng sau”. Ngày không ghi năm được hiểu là lần xuất hiện gần nhất sắp tới.
  \item Giờ được chuyển sang hệ 24 giờ theo Bảng~\ref{tab:hours}; kết quả \textit{không phụ thuộc giờ hiện tại}, nên cùng một câu luôn cho cùng một kết quả.
\end{itemize}

\subsection{Chính sách khi tạo sự kiện}
\begin{itemize}
  \item Không có cả ngày lẫn giờ: hỏi lại (ví dụ “Bạn muốn đặt lịch \ldots{} vào lúc nào?”).
  \item Chỉ có ngày: đặt 9 giờ sáng ngày đó nếu thời điểm này còn ở phía trước, nếu không thì hỏi giờ.
  \item Thời lượng mặc định 60 phút; các cuộc gọi, cà phê, đồng bộ nhanh mặc định 30 phút; nếu người dùng nêu giờ kết thúc hoặc thời lượng thì dùng giá trị đó.
  \item Ngày rơi vào quá khứ (ví dụ nói “thứ hai này” vào thứ tư): hỏi lại người dùng muốn ngày nào, không tạo sự kiện trong quá khứ.
  \item Thiếu tiêu đề: dùng tiêu đề mặc định theo ngôn ngữ của người dùng.
\end{itemize}

\subsection{Xác định sự kiện mục tiêu khi sửa và xóa}
Sửa và xóa đòi hỏi tìm đúng \textit{một} sự kiện hiện có. Bộ tìm mục tiêu hoạt động như sau:
\begin{enumerate}
  \item \textbf{Xác định cửa sổ tìm kiếm:} ±120 phút quanh thời điểm người dùng nêu rõ; nếu không thì ngày, buổi hoặc khoảng thời gian được nêu; nếu không có thông tin thời gian thì 30 ngày tới tính từ hiện tại. Không bao giờ quét về quá khứ xa để lấy “sự kiện cũ nhất”.
  \item \textbf{Chấm điểm ứng viên:} theo số từ nội dung của cụm nhận diện xuất hiện trong tiêu đề (trọng số 1), địa điểm (0,8), mô tả (0,6), cộng độ gần về thời gian. Dấu tiếng Việt được so sánh khi cả hai phía đều có dấu và bỏ qua khi một phía không dấu; các từ chung chung như “cuộc”, “họp”, “meeting” chỉ dùng để xếp hạng.
  \item \textbf{Quyết định:} chỉ tác động khi có đúng một ứng viên khớp đầy đủ và cách biệt rõ với ứng viên kế tiếp. Nếu hai ứng viên có điểm sát nhau (chênh dưới 0,15), hoặc người dùng chỉ nêu ngày, hệ thống liệt kê tối đa năm ứng viên để người dùng chọn. Khớp một phần hoặc chỉ khớp khi bỏ dấu thì chỉ được \textit{đề xuất}, không tự thay đổi. Không có tên lẫn thời gian thì hỏi lại; không tìm thấy thì báo rõ.
\end{enumerate}
Kế hoạch cập nhật giữ nguyên thời lượng trừ khi người dùng nêu giờ kết thúc; đích chỉ có giờ thì giữ nguyên ngày của sự kiện; hỗ trợ dịch tương đối; phát hiện khoảng thời gian không hợp lệ.

\section{Quản lý hội thoại và xác nhận thao tác}\label{sec:dialog-design}
Mỗi câu lệnh thuộc về một \textit{hội thoại}. Nếu yêu cầu không kèm mã hội thoại, hệ thống tạo hội thoại mới với tiêu đề lấy từ câu đầu tiên. Tin nhắn của người dùng, câu trả lời của trợ lý và thao tác chờ (nếu có) được ghi trong \textit{một giao dịch} cơ sở dữ liệu dưới khóa hàng của hội thoại, bảo đảm thứ tự nhất quán khi có yêu cầu đồng thời. Ngôn ngữ trả lời được chọn theo thứ tự: tham số của yêu cầu → cài đặt người dùng → ngôn ngữ phát hiện từ câu lệnh.

\begin{itemize}
  \item \textbf{Một thao tác chờ cho mỗi hội thoại:} đề xuất mới làm hết hiệu lực đề xuất cũ; lời xác nhận chỉ có giá trị trong chính hội thoại đã hỏi.
  \item \textbf{Thực thi đúng một lần:} việc chuyển trạng thái \textit{pending} sang \textit{confirmed}/\textit{rejected} là một phép so-sánh-rồi-gán nguyên tử trong cơ sở dữ liệu và chỉ thành công khi còn hạn; hai lời xác nhận đồng thời chỉ dẫn tới một lần ghi lên lịch.
  \item \textbf{Kiểm tra lại trước khi ghi:} khi xác nhận, sự kiện được đọc lại; nếu sự kiện không còn tồn tại (hoặc người dùng đã đổi nguồn lịch) thì báo rõ và không thực hiện.
  \item \textbf{Phục hồi khi lỗi:} nếu việc ghi lên lịch thất bại, thao tác được mở lại để người dùng thử lại, ví dụ sau khi đăng nhập Google lại.
  \item \textbf{Hai cách xác nhận tương đương:} bấm nút (gọi API xác nhận theo mã thao tác) hoặc trả lời bằng lời/gõ “có”, “không” trong cùng hội thoại.
\end{itemize}
Lịch sử hội thoại được truy vấn theo phân trang con trỏ (keyset), mới nhất trước; xóa hội thoại sẽ làm hết hiệu lực các thao tác chờ của nó.

\section{Thiết kế phân hệ giọng nói}
\subsection{Nhận dạng tiếng nói}
Bộ STT được chọn bằng biến \ma{STT_PROVIDER}: \textit{gemini}, \textit{mock} (chỉ cho phát triển và kiểm thử, bị cấm ở production) hoặc \textit{none}. Yêu cầu nhận dạng được kiểm tra theo thứ tự: bộ STT khả dụng (nếu không trả 503) → âm thanh hợp lệ. Âm thanh rỗng trả lỗi 400; vượt 10 MB sau giải mã trả 413; kiểu tệp được chuẩn hóa và nếu thiếu thì được đoán từ các byte đầu tệp; chấp nhận WAV, WebM, Ogg, MP3, MP4 và FLAC, còn lại trả 415. Tệp WAV được phân tích tiêu đề để tính thời lượng.

Bộ STT Gemini gửi âm thanh kèm lời nhắc yêu cầu chép nguyên văn và \textit{không dịch}, nhiệt độ 0; toàn bộ các lần thử (kể cả mô hình dự phòng) dùng chung một ngân sách 30 giây. Đoạn im lặng cho kết quả rỗng và không kích hoạt agent. Khác với nguyên mẫu ban đầu, khi STT lỗi hệ thống \textit{không} chuyển sang bộ giả lập mà trả lỗi rõ ràng.

\subsection{Tổng hợp tiếng nói}
Giọng đọc được chọn theo thứ tự: giọng nêu trong yêu cầu → giọng người dùng đã lưu → giọng mặc định của máy chủ → giọng tiếng Việt đầu tiên tìm thấy. Văn bản được làm sạch các ký hiệu định dạng trước khi đọc và giới hạn 1000 ký tự mỗi yêu cầu (ứng dụng tự chia nhỏ ở mức 900 ký tự).

Piper được chạy như tiến trình con với các biện pháp an toàn: không qua shell; văn bản truyền qua luồng nhập chuẩn chứ không qua tham số dòng lệnh; biến môi trường theo danh sách cho phép; mã giọng đọc chỉ được so khớp với các mô hình đã quét thấy trong thư mục mô hình (không biến dữ liệu từ client thành đường dẫn); tệp kết quả tạm có tên ngẫu nhiên và luôn bị xóa; quá 15 giây thì tiến trình bị buộc dừng; số tiến trình đồng thời bị giới hạn bằng semaphore (mặc định 2). Khi TTS không khả dụng, API trả 503 thay vì phát âm thanh thay thế.

Các quy tắc của giao thức:
\begin{itemize}
  \item \textbf{Xác thực:} bằng tiêu đề khi nâng cấp kết nối hoặc khung \ma{auth} đầu tiên trong vòng 10 giây. Trước khi xác thực, tầng truyền chỉ nhận khung tối đa 16 KiB, nên client phải chờ \ma{ready} rồi mới gửi âm thanh. Khi token hết hạn giữa chừng, yêu cầu mới bị từ chối bằng khung lỗi nhưng kết nối được giữ để client gửi lại \ma{auth}.
  \item \textbf{Một yêu cầu tại một thời điểm} trên mỗi kết nối; yêu cầu thứ hai nhận lỗi \ma{BUSY}. \ma{cancel} làm máy chủ gửi ngay trạng thái \textit{idle} và bỏ các khung còn lại của yêu cầu.
  \item \textbf{Tách câu:} câu trả lời được tách thành từng câu, giữ nguyên giờ, số thập phân, ngày tháng, URL và chữ viết tắt; mỗi dòng của danh sách là một khung riêng. Lỗi tổng hợp giọng chỉ sinh một khung lỗi; các câu còn lại vẫn được gửi dạng văn bản.
  \item \textbf{Giới hạn:} 60 khung yêu cầu mỗi phút cho mỗi người dùng, riêng lệnh giọng nói 30 khung mỗi phút; khung sau xác thực tối đa khoảng 13,4 MB với cấu hình mặc định; máy chủ gửi ping giao thức mỗi 30 giây và ngắt kết nối không phản hồi.
\end{itemize}
Điểm cuối \texttt{POST /voice/command} cung cấp cùng chuỗi xử lý (nhận dạng → agent → giọng đọc) trong một yêu cầu HTTP cho các client không dùng WebSocket. Ở thời điểm báo cáo, kênh WebSocket đã được hiện thực và kiểm thử ở backend; việc chuyển ứng dụng desktop từ ba lời gọi REST sang kênh này nằm trong kế hoạch giai đoạn cuối.
